### 3b — Add _tick_hrcw_checkbox function

```python
def _tick_hrcw_checkbox(cell, keyword: str) -> bool:
    """
    Find the checkbox (☐) adjacent to `keyword` in the merged HRCW cell
    and replace with ticked (☑). Returns True if ticked, False if not found.

    New template places all HRCW checkboxes inline in one merged cell.
    Cell-address targeting no longer works — must target by keyword proximity.
    Inference pipeline output is unchanged. Only render target changes.
    """
    keyword_lower = keyword.lower()
    for para in cell.paragraphs:
        if keyword_lower not in para.text.lower():
            continue
        for run in para.runs:
            if "☐" in run.text:
                run.text = run.text.replace("☐", "☑", 1)
                run.bold = True
                return True
    return False
```

```python
def _build_task_table(doc, tasks) -> None:
    """Populate T1 — 8-col step table. Banner row (0) untouched. Headers at row 1."""
    t1 = doc.tables[1]
    _set_table_cell_margins(t1)

    # Keep row 0 (banner) and row 1 (headers). Delete all sample data rows.
    while len(t1.rows) > 2:
        tr = t1.rows[2]._tr
        tr.getparent().remove(tr)

    # Reformat row 1 column headers. Row 0 is black banner — leave untouched.
    for i, h in enumerate(_HEADERS):
        cell = t1.rows[1].cells[i]
        for para in cell.paragraphs:
            para.clear()
        _header_cell(cell, h, size_pt=_SZ)

    # Add one row per task
    for idx, task in enumerate(tasks, 1):
        row = t1.add_row()
        c = row.cells
        for i, w in enumerate(_COL_W_DXA):
            c[i].width = w

        # Col 0 — Step number
        _run(c[0].paragraphs[0], str(idx), bold=True, size_pt=_SZ)

        # Col 1 — Task name + scope
        _run(c[1].paragraphs[0], task.task, bold=True, size_pt=_SZ)
        if task.scope:
            _run(c[1].add_paragraph(), "[%s]" % task.scope, color=GREY, size_pt=_SZ)

        # Col 2 — Hazards
        _hazard_cell(c[2], task, size_pt=_SZ)

        # Col 3 — Risk Pre (10pt bold per Gatekeeper standard)
        _risk_cell(c[3], task.risk_pre or "", size_pt=10)

        # Col 4 — Controls
        _controls_cell(c[4], task, size_pt=_SZ)

        # Col 5 — Risk Post (10pt bold per Gatekeeper standard)
        _risk_cell(c[5], task.risk_post or "", size_pt=10)

        # Col 6 — Responsible
        _responsibility_cell(c[6], task, size_pt=_SZ)

        # Col 7 — Hold Points + Stop-Work Triggers + CCVS code
        _hold_point_cell(c[7], task, size_pt=_SZ)
```

```python
def _hold_point_cell(cell, task, size_pt: int = 9) -> None:
    """
    Render hold points, stop-work triggers, and CCVS code into col 7.
    ⚠️ HOLD POINT  — bold, colour #2E75B6
    🛑 STOP-WORK TRIGGER — bold, colour #C00000
    CCVS code — bold grey, appended at bottom beneath stop-work triggers.
    """
    p = cell.paragraphs[0]
    first = True

    if task.hold_points:
        for item in task.hold_points:
            if not first:
                p = cell.add_paragraph()
            text = item.lstrip("⚠️").strip()
            _run(p, "⚠️ HOLD POINT", bold=True, color=BLUE_HOLD, size_pt=size_pt)
            if text:
                _run(p, f"\n{text}", size_pt=size_pt)
            first = False

    if task.stop_work:
        for item in task.stop_work:
            if not first:
                p = cell.add_paragraph()
            text = item.lstrip("🛑").strip()
            _run(p, "🛑 STOP-WORK TRIGGER", bold=True, color=RED, size_pt=size_pt)
            if text:
                _run(p, f"\n{text}", size_pt=size_pt)
            first = False

    if first:  # nothing written yet
        _run(p, "—", size_pt=size_pt)

    # CCVS code — appended at bottom of cell, bold grey
    code = validate_ccvs_code(task.ccvs_code or "N/A")
    if code != "N/A":
        p = cell.add_paragraph()
        _run(p, code, bold=True, color=GREY, size_pt=size_pt)
```

```python
def _build_monitoring_table(doc, tasks) -> None:
    """
    Populate T2 — monitoring rows.
    One row per task with a valid CCVS code.
    If monitoring object is null, render dash placeholders — do not skip row.
    CCVS code renders in step table col 7 only — not here.
    """
    t2 = doc.tables[2]
    _set_table_cell_margins(t2)

    # Reformat header row
    for i, h in enumerate(_MON_HEADERS):
        cell = t2.rows[0].cells[i]
        for para in cell.paragraphs:
            para.clear()
        _header_cell(cell, h, size_pt=_SZ)

    # Remove blank template data row (row 1), keep header only
    if len(t2.rows) > 1:
        tr = t2.rows[1]._tr
        tr.getparent().remove(tr)

    # Add one row per task — skip only if CCVS code is N/A
    for task in tasks:
        code = validate_ccvs_code(task.ccvs_code or "N/A")
        if code == "N/A":
            continue  # no valid CCVS code — no monitoring row

        m = task.monitoring  # may be None — render dashes if so
        row = t2.add_row()
        for i, w in enumerate(_MON_W):
            row.cells[i].width = Cm(w)

        vals = [
            task.task,
            m.critical_control if m else "—",
            m.who              if m else "—",
            m.frequency        if m else "—",
            m.evidence         if m else "—",
        ]
        for i, val in enumerate(vals):
            p = row.cells[i].paragraphs[0]
            _run(p, val or "—", size_pt=_SZ)
```

```python
def _fill_prerequisites_table(doc, tasks, inference, project_meta, jur, jurisdiction) -> None:
    """
    Populate T9 — 8-cell pre-requisites table.
    Replaces _fill_legislation_table and _fill_requirements_table.
    All assembly logic moved here verbatim — only cell destination changes.
    Content rules: max 6 items per cell, concise format per Gatekeeper standard.
    """
    t9 = doc.tables[9]
    _set_table_cell_margins(t9)

    def _set_prereq(row_idx: int, col_idx: int, items: list[str]) -> None:
        """Write bullet list into a T9 cell. Max 6 items."""
        cell = t9.cell(row_idx, col_idx)
        for para in cell.paragraphs:
            para.clear()
        if not items:
            _run(cell.paragraphs[0], "—", size_pt=_SZ)
            return
        capped = items[:6]  # hard cap — field document must be scannable
        p = cell.paragraphs[0]
        for i, item in enumerate(capped):
            if i > 0:
                p = cell.add_paragraph()
            _run(p, f"•  {item}", size_pt=_SZ)

    # ── (1,1) PPE — All Persons ───────────────────────────────────────────────
    # Move existing PPE assembly logic verbatim from old _fill_requirements_table
    # (baseline list + inference["ppe"] + dedup). Output → final_ppe.
    # ... [existing PPE assembly logic] ...
    _set_prereq(1, 1, final_ppe)

    # ── (1,3) Additional PPE — WAH ───────────────────────────────────────────
    wah_ppe = [
        "Full body harness (AS/NZS 1891.1) with rope grab",
        "Trauma straps fitted to all harnesses",
        "P2 respirator (AS/NZS 1716)",
        "Hearing protection if ≥85 dBA",
        "Sun protection SPF 50+",
        "Hard hat with chin strap for elevated work",
    ]
    _set_prereq(1, 3,
        wah_ppe if any(t.wah_applicable for t in tasks)
        else ["As required for task — see controls column"])

    # ── (2,1) Licences / Qualifications ──────────────────────────────────────
    quals = inference.get("qualifications", []) + inference.get("certifications", [])
    _set_prereq(2, 1, quals)

    # ── (2,3) Permits & Approvals ────────────────────────────────────────────
    _set_prereq(2, 3, inference.get("permits", []))

    # ── (3,1) Plant & Equipment ──────────────────────────────────────────────
    # Move existing plant assembly logic verbatim from old row 4.
    # Trim verbose maintenance text to: "{name} — pre-start checklist required"
    # ... [existing plant assembly logic, output → plant_items] ...
    _set_prereq(3, 1, plant_items)

    # ── (3,3) Hazardous Substances ───────────────────────────────────────────
    # Move existing haz_sub assembly logic verbatim from old row 6.
    # Trim to: "{name} — SDS on site — {one control note}"
    # Remove WES limits, RPE spec — those belong in controls column.
    # ... [existing haz_sub assembly logic, output → haz_sub_items] ...
    _set_prereq(3, 3, haz_sub_items)

    # ── (4,1) Consultation ───────────────────────────────────────────────────
    consultation = [
        "Workers consulted per WHS Reg r.299(3) and r.300(2)",
        "SWMS reviewed with all workers before work starts",
        "Workers sign off before starting",
    ]
    _set_prereq(4, 1, consultation)

    # ── (4,3) Legislative Basis ──────────────────────────────────────────────
    # Move existing legislation assembly verbatim from old _fill_legislation_table.
    # Trim to primary acts only — max 4 items total.
    # Codes of Practice → one line "Applicable Codes of Practice".
    # ... [existing legislation assembly, output → all_legislation] ...

```
